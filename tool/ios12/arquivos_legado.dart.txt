import 'dart:typed_data';

import 'package:file_picker/file_picker.dart';

/// Versão para o file_picker 10 (Flutter 3.32 / iOS 12).
Future<String?> salvarArquivo({required String titulo, required String nome, required Uint8List bytes}) =>
    FilePicker.platform.saveFile(dialogTitle: titulo, fileName: nome, bytes: bytes);

Future<Uint8List?> escolherArquivo() async {
  final r = await FilePicker.platform.pickFiles(withData: true);
  if (r == null || r.files.isEmpty) return null;
  return r.files.single.bytes;
}
